"""Inert motor-carriage candidate, all dimensions IP-local millimetres.

Only build(cq, original_motor_plate_IP_local) constructs shapes. No import-time
CAD, file access, export, process, or repository modification. The caller owns
saved-input hashing, common-plate drilling, inherited-row placement, and audit.
This module proposes geometry and required checks; it contains no pass claims.
"""
from __future__ import annotations

import math


LOCK_HOLES = ((-50., -156.), (8., -156.), (-50., -99.), (8., -99.))
SHOE_HOLES = ((-62.5, -156.), (20.5, -156.), (-62.5, -99.), (20.5, -99.))
MOTOR_PLATE = 'drive_bench_baseplate'
COMMON = 'bench_common_plate'
TRAVEL = (-2., 2.)
PLATE_BOUNDS = (-56., -162.5, -21.8, 14., -92.5, -18.8)
WASHER_T = .9  # Accu HPW-M4-A2 published maximum, actual range .7.. .9.
HARDWARE = dict(screw='Accu SSC-M4-25-A2', nut='Accu HPN-M4-A2',
    washer='Accu HPW-M4-A2', screw_length_mm=25., shaft_diameter_mm=4.,
    head_diameter_mm=7., head_height_mm=4., socket_across_flats_mm=3.,
    socket_depth_mm=2., pitch_mm=.7, nut_across_flats_mm=7., nut_height_mm=3.2,
    washer_outer_diameter_mm=9., washer_inner_diameter_mm=4.3,
    washer_thickness_mm=.9, washer_thickness_range_mm=[.7, .9],
    threads='Smooth major-diameter envelopes; no helical or incomplete threads.',
    source_binding='Caller must bind unchanged thumb_ip_bench_v1/source_docs originals.')
SOCKET_URL = 'https://hybris-media.wera.de/download/pdfgenerator-datasheets/en/05004504001.pdf'


def build(cq, original_motor_plate_IP_local):
    """Return new shapes and explicit changes to the 124-part saved bench.

    Input is the bench's already drilled 70x70x3 motor plate, in IP coordinates.
    The plate is replaced; its four old spacers and four upper washers are
    removed. Other motor geometry is not reconstructed. Four inherited M4x25
    locking screws keep their native bytes and get a +1.1 mm Z row update.
    The two shoes are independently fixed by four NEW M4x25 stacks; four common
    plate holes are returned as requests, not silently cut into another part.
    The adjuster module alone owns the forward tab, nut pocket and capture cap.
    """
    from OCP.BRepAlgoAPI import BRepAlgoAPI_Cut, BRepAlgoAPI_Fuse
    from OCP.BRepClass3d import BRepClass3d_SolidClassifier
    from OCP.TopAbs import TopAbs_OUT
    from OCP.TopTools import TopTools_ListOfShape

    parts, contacts, assembly = [], [], []
    names = set()

    def facts(shape):
        solids = shape.Solids()
        bounded = True
        for solid in solids:
            classifier = BRepClass3d_SolidClassifier(solid.wrapped)
            classifier.PerformInfinitePoint(1e-7)
            bounded = bounded and classifier.State() == TopAbs_OUT
        return dict(valid=bool(shape.isValid()), solids=len(solids),
            positive=bool(solids) and all(s.Volume() > 1e-7 for s in solids),
            volume_mm3=sum(s.Volume() for s in solids), bounded=bool(bounded))

    def require_solid(shape, label):
        f = facts(shape)
        if not (f['valid'] and f['solids'] == 1 and f['positive'] and f['bounded']):
            raise ValueError('Invalid/disconnected/inverted '+label+': '+str(f))
        return f

    def boolean(kind, left, *right):
        old = [s.Volume() for s in (left, *right)]
        operation = {'cut': BRepAlgoAPI_Cut, 'fuse': BRepAlgoAPI_Fuse}[kind]()
        aa, bb = TopTools_ListOfShape(), TopTools_ListOfShape()
        aa.Append(left.copy().wrapped)
        for shape in right:
            bb.Append(shape.copy().wrapped)
        operation.SetArguments(aa); operation.SetTools(bb)
        operation.SetNonDestructive(True); operation.SetRunParallel(False)
        operation.Build()
        if not operation.IsDone():
            raise ValueError('Unfinished '+kind)
        result = cq.Shape.cast(operation.Shape())
        require_solid(result, kind+' result')
        if any(abs(s.Volume()-v) > 1e-8 for s, v in zip((left, *right), old)):
            raise ValueError('Boolean mutated an input')
        if kind == 'cut' and result.Volume() > old[0]+1e-6:
            raise ValueError('Cut added volume')
        return result

    def box(x0, x1, y0, y1, z0, z1):
        return cq.Solid.makeBox(x1-x0, y1-y0, z1-z0, cq.Vector(x0, y0, z0))

    def cylinder(x, y, radius, z0, z1):
        return cq.Solid.makeCylinder(radius, z1-z0, cq.Vector(x, y, z0), cq.Vector(0, 0, 1))

    def hex_prism(x, y, af, z0, z1):
        radius = af/math.sqrt(3.)
        points = [(x+radius*math.cos(i*math.pi/3.), y+radius*math.sin(i*math.pi/3.)) for i in range(6)]
        return cq.Workplane('XY').workplane(offset=z0).polyline(points).close().extrude(z1-z0, clean=False).val()

    def ring(x, y, z0, z1):
        return boolean('cut', cylinder(x, y, 4.5, z0, z1), cylinder(x, y, 2.15, z0-.01, z1+.01))

    def screw(x, y):
        shape = boolean('fuse', cylinder(x, y, 2., -42.9, -17.9),
                        cylinder(x, y, 3.5, -17.9, -13.9))
        return boolean('cut', shape, hex_prism(x, y, 3., -15.9, -13.89))

    def add(name, shape, role, body='fixed', color=(.46, .55, .65)):
        if name in names:
            raise ValueError('Duplicate part: '+name)
        names.add(name)
        require_solid(shape, name)
        parts.append(dict(name=name, shape=shape, body=body, role=role, color=list(color)))
        return name

    def contact(a, b, area, description, z, xy, radius=None, patch=None, frame='fixed'):
        row = dict(a=a, b=b, minimum_area_mm2=float(area), description=description,
            plane_origin_mm=[float(xy[0]), float(xy[1]), float(z)], normal=[0., 0., 1.],
            contact_frame=frame)
        if radius is not None:
            row['patch_radius_mm'] = float(radius)
        if patch is not None:
            row['patch_box_mm'] = [[float(patch[0]), float(patch[1]), float(z)-.1],
                                   [float(patch[2]), float(patch[3]), float(z)+.1]]
        contacts.append(row)

    def screw_access(name, x, y, z, description):
        assembly.append(dict(kind='axial_screw', part=name, underhead=[x, y, z], axis=[0., 0., 1.],
            shaft_radius_mm=2., head_radius_mm=3.5, head_height_mm=4., length_mm=25.,
            entry_travel_mm=35., driver_radius_mm=2., driver_length_mm=40.,
            stage_excluded=[], access_frame='fixed', description=description))

    original_facts = require_solid(original_motor_plate_IP_local, 'input motor plate')
    bb = original_motor_plate_IP_local.BoundingBox()
    got = (bb.xmin, bb.ymin, bb.zmin, bb.xmax, bb.ymax, bb.zmax)
    if any(abs(a-b) > .02 for a, b in zip(got, PLATE_BOUNDS)):
        raise ValueError('Input plate is not in the expected IP-local frame: '+str(got))
    plate = original_motor_plate_IP_local.copy()
    for x, y in LOCK_HOLES:
        slot = cq.Workplane('XY').workplane(offset=-21.82).center(x, y).slot2D(8.5, 4.5, 90.).extrude(3.04, clean=False).val()
        plate = boolean('cut', plate, slot)
    add(MOTOR_PLATE, plate, 'copied_motor_plate_four_Y_adjustment_slots', 'carriage')

    shoes = {}
    for side, x0, x1, wall0, wall1, lip0, lip1, cx in (
            ('left', -68., -44., -68., -56.15, -68., -54.5, -62.5),
            ('right', 2., 26., 14.15, 26., 12.5, 26., 20.5)):
        # Solid lower bearing surface, outside guide wall, then fixed lift lip.
        # A 0.3 mm vertical gap remains over the 3 mm moving plate under the lip.
        shoe = boolean('fuse', box(x0, x1, -165., -90., -28., -21.8),
            box(wall0, wall1, -165., -90., -21.8, -18.5),
            box(lip0, lip1, -146., -109., -18.5, -16.5))
        lock_x = -50. if side == 'left' else 8.
        for y in (-156., -99.):
            shoe = boolean('cut', shoe, cylinder(lock_x, y, 2.25, -28.02, -21.78))
            shoe = boolean('cut', shoe, cylinder(cx, y, 2.25, -28.02, -18.48))
        sn = add('tension_carrier_'+side+'_shoe', shoe, 'independently_fixed_continuous_support_X_guide_and_lift_lip')
        shoes[side] = sn
        contact(sn, COMMON, 1500., 'Continuous lower shoe footprint; independent bolts retain its datum during adjustment',
            -28., ((x0+x1)/2., -127.5), patch=[x0, -165., x1, -90.])
        contact(sn, MOTOR_PLATE, 600., 'Moving plate underside supported across the complete travel; no friction/strength approval',
            -21.8, (lock_x, -127.5), patch=[-56. if side == 'left' else 2., -165., -44. if side == 'left' else 14., -90.])

    # Wall upper seating is -18.5; local pads at new screw stations are reduced
    # to -18.8 to retain the old verified M4x25/0.9 mm washer stack height.
    for p in parts:
        if p['name'] in shoes.values():
            cx = -62.5 if p['name'] == shoes['left'] else 20.5
            shape = p['shape']
            for y in (-156., -99.):
                shape = boolean('cut', shape, box(cx-5., cx+5., y-5., y+5., -18.8, -18.49))
            p['shape'] = shape

    for i, ((x, y), (fx, fy)) in enumerate(zip(LOCK_HOLES, SHOE_HOLES), 1):
        # The bridge spans both sides of the slot and has much more seating area
        # than the removed round washer. It locks to the separately fixed shoe.
        clamp = boolean('cut', box(x-6., x+6., y-8., y+8., -18.8, -16.8),
                        cylinder(x, y, 2.25, -18.82, -16.78))
        cn = add('tension_carrier_lock_bridge_'+str(i), clamp, 'custom_2mm_bridge_clamp_over_adjustment_slot')
        inherited = 'bench_motor_mount_'+str(i)+'_screw'
        contact(cn, MOTOR_PLATE, 80., 'Bridge seats on both slot banks at each audited carriage displacement',
            -18.8, (x, y), patch=[x-6., y-8., x+6., y+8.])
        contact(inherited, cn, 20., 'Retained M4x25 head underside on bridge; screw has only a placement update',
            -16.8, (x, y), radius=3.5)
        screw_access(inherited, x, y, -16.8,
            'Install after carrier slides into fixed shoes. Loosen these four locks for adjustment; separate shoe bolts stay tight. Insert/drive from +Z, then re-lock before any actuation.')
        prefix = 'tension_carrier_shoe_mount_'+str(i)
        fs = add(prefix+'_screw', screw(fx, fy), 'M4x25_supplier_dimension_major_envelope')
        uw = add(prefix+'_upper_washer', ring(fx, fy, -18.8, -17.9), 'M4_washer_supplier_maximum_0p9mm')
        lw = add(prefix+'_lower_washer', ring(fx, fy, -36.9, -36.), 'M4_washer_supplier_maximum_0p9mm')
        nt = add(prefix+'_nut', boolean('cut', hex_prism(fx, fy, 7., -40.1, -36.9),
            cylinder(fx, fy, 2., -40.12, -36.88)), 'M4_nut_supplier_major_envelope')
        sn = shoes['left' if fx < 0. else 'right']
        contact(fs, uw, 22., 'Shoe fixing head on upper washer', -17.9, (fx, fy), radius=3.5)
        contact(uw, sn, 45., 'Upper washer on locally lowered shoe seat', -18.8, (fx, fy), radius=4.5)
        contact(lw, COMMON, 45., 'Shoe fixing lower washer against common underside', -36., (fx, fy), radius=4.5)
        contact(nt, lw, 26., 'Shoe fixing nut on lower washer', -36.9, (fx, fy), radius=4.5)
        screw_access(fs, fx, fy, -17.9,
            'Independently secure the empty guide shoe to common plate before carrier insertion; these bolts stay tight during Y adjustment.')
        assembly.append(dict(kind='axial_nut', part=nt, mate_screw=fs, axis=[0., 0., -1.],
            upper_face=[fx, fy, -36.9], across_flats_mm=7., height_mm=3.2,
            entry_travel_mm=30., path_travel_mm=30., socket_max_dia_mm=11.8,
            socket_length_mm=50., socket_part_number='Wera 05004504001', source_url=SOCKET_URL,
            stage='before_table_placement', stage_excluded=[nt, fs], access_frame='fixed',
            description='Actual nut insertion and conservative exterior socket access still require native audit; internal tool bore/extension and tightening remain unqualified.'))

    removed = [f'bench_motor_mount_{i}_{suffix}' for i in range(1, 5) for suffix in ('spacer', 'upper_washer')]
    lock_screws = [f'bench_motor_mount_{i}_screw' for i in range(1, 5)]
    if len(parts) != 23 or len(contacts) != 28 or len(assembly) != 12:
        raise ValueError('Incomplete carrier declarations')
    if facts(original_motor_plate_IP_local) != original_facts:
        raise ValueError('Original motor plate was changed')
    return dict(parts=parts, remove_names=removed, replace_names=[MOTOR_PLATE],
        row_updates={name: dict(body='fixed', neutral_translation_delta_mm=[0., 0., 1.1]) for name in lock_screws},
        remove_contact_parts=removed, replace_assembly_parts=lock_screws,
        contacts=contacts, assembly=assembly,
        required_common_holes=[dict(xy=list(xy), diameter=4.5, diameter_mm=4.5,
            span_z_mm=[-36., -28.], purpose='independent carrier shoe attachment') for xy in SHOE_HOLES],
        spec=dict(schema='thumb_ip_adjustable_carrier.v1', units='IP-local millimetres',
            source_motor_plate_name=MOTOR_PLATE, original_plate_facts=original_facts,
            original_motor_geometry_other_than_plate_preserved=True,
            hand_and_parent_fixture_geometry_changed=False,
            travel_y_mm=list(TRAVEL), axis_spacing_mm_at_neutral=120.,
            spacing_rule='D_mm = 120 - carriage_translation_y_mm; D118/120/122 use Y+2/0/-2.',
            moving_group='Motor plate plus all original 32 motor components; original component bytes remain unchanged, caller applies carrier translation before motor output rotation.',
            locked_nominal_contacts=True, guide_inner_x_mm=[-56.15, 14.15],
            guide_side_clearance_mm=.15,
            ideal_rectangle_side_guide_yaw_limit_deg=math.degrees(math.asin(70.3/(70.*math.sqrt(2.)))-math.pi/4.),
            lift_lip_bottom_z_mm=-18.5, lift_gap_mm=.3,
            lift_lip_y_mm=[-146., -109.], support_top_z_mm=-21.8,
            independent_shoe_mount_holes_xy_mm=[list(v) for v in SHOE_HOLES],
            lock_holes_xy_mm=[list(v) for v in LOCK_HOLES],
            slot_length_width_mm=[8.5, 4.5], slot_centerline_travel_mm=4.,
            lock_bridge_thickness_mm=2., lock_screw_underhead_z_mm=-16.8,
            lock_screw_tip_z_mm=-41.8, retained_nut_span_z_mm=[-40.1, -36.9],
            lock_screw_nominal_tip_projection_mm=1.7,
            independent_shoe_screw_underhead_z_mm=-17.9,
            independent_shoe_screw_tip_z_mm=-42.9,
            hardware_assumptions=dict(HARDWARE),
            adjuster_interface=dict(owner='adjuster.py',
                carrier_plate_key='parts row whose name is drive_bench_baseplate',
                plate_front_edge_y_mm=-92.5, plate_span_x_mm=[-56., 14.],
                plate_span_z_mm=[-21.8, -18.8],
                proposed_tab_bounds_mm=[-31., -94., -26., -11., -86., -17.],
                cap_highest_z_mm=-15., screw_axis_xz_mm=[-21., -20.5],
                tab_generated_by_this_module=False,
                function_contract='adjuster.build(cq, carrier_plate) returns modified carrier_plate and its distinct parts/metadata; caller replaces this module plate shape once.',
                status='Interface agreed; actual cord/tool/other-part clearance unverified.'),
            carrier_insertion=dict(direction=[0., 1., 0.], start_translation_mm=[0., -73., 0.],
                end_translation_mm=[0., 0., 0.],
                parts_absent=lock_screws+[f'tension_carrier_lock_bridge_{i}' for i in range(1, 5)],
                adjuster_and_parent_coupon_not_yet_installed=True,
                description='Bench preassembly: slide motor/plate from rear -Y into already fixed shoes before adding lock bridges, bolts and adjuster. This is an untested required assembly path, not included among the axial access declarations.'),
            pending_checks=[
                'Common four new holes and their native wall/seat/bolt/nut/tool access.',
                'All new/retained native parts, cord segments and clearances at D118/120/122; intermediate Y travel and motor rotation need their declared sampling/sweeps.',
                'All 28 local seats at the three carriage displacements; slot banks must carry bridge clamps and lower rails.',
                'Rear carrier insertion including full 32-part motor module, fixed lips and rail fasteners; adjuster/tab construction order.',
                'Guide clearance, yaw/lift free play, hole tolerances, binding, wear, friction and clamp preload/strength.',
                'Two-millimetre custom bridge material and bending strength, screw engagement and loosening; geometric seats alone do not prove locking capacity.',
                'Adjuster tab/nut/cap highest Z-15 and actual cord/tool clearance after the adjuster module is applied.'
            ], manufacturing_approved=False, assembly_verified=False,
            load_or_tendon_preload_approved=False, measured_latency=False))
