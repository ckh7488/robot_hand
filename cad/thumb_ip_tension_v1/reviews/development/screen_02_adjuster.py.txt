"""Inert, untested IP-local tension-stage adjuster construction.

build(cq, carrier_plate) accepts the independently slotted motor plate. It returns
its modified copy and separate fixed/carriage/adjuster_rotor solids. Import does
not load CAD, write files, launch processes, or run geometry. All dimensions are
millimetres. The caller must perform saved-BRep validation, intersections, finite
seats, insertion, tools, and tolerance checks before accepting this candidate.

This is a captured-head RH M4 screw with a keyed travelling nut, NOT a spring
preload regulator. Separate carrier clamps are required for service operation.
Threads are major-diameter envelopes; no force or physical approval is implied.
"""
from __future__ import annotations

import math


P = dict(
    axis_x_mm=-21., axis_z_mm=-20.5,
    carriage_range_y_mm=[-2., 2.], nominal_axis_spacing_mm=120.,
    screw_underhead_y_mm=-73., screw_length_mm=25., screw_pitch_mm=.7,
    signed_travel_mm_per_revolution=-.7,
    head_diameter_mm=7., head_height_mm=4., head_socket_af_mm=3.,
    head_socket_depth_mm=2., shaft_diameter_mm=4.,
    washer_od_mm=9., washer_id_mm=4.3, washer_max_thickness_mm=.9,
    washer_min_thickness_mm=.7, head_min_height_mm=3.82,
    head_capture_play_mm=.1, fixed_thrust_y_mm=-73.9,
    front_cap_rear_y_mm=-68.9, front_cap_front_y_mm=-64.9,
    front_cap_driver_radius_mm=2.1,
    cap_bolt_x_mm=[-36.5, -5.5], cap_bolt_length_mm=20.,
    fixed_post_rear_y_mm=-81.,
    tab_bounds_mm=[[-31., -94., -26.], [-11., -86., -17.]],
    nut_af_mm=7., nut_max_height_mm=3.2, nut_min_height_mm=2.9,
    nut_y_mm=[-91.3, -88.1], nut_pocket_y_mm=[-91.3, -87.9],
    nut_pocket_af_mm=7.2, nut_loading_halfwidth_mm=4.2,
    nut_cap_bounds_mm=[[-31., -94., -17.], [-11., -86., -15.]],
    nut_cap_roof_clearance_mm=.1,
    nut_cap_screw_x_mm=[-28., -14.], nut_cap_screw_y_mm=-89.6,
    nut_cap_screw_total_length_mm=6., nut_cap_screw_head_diameter_mm=4.7,
    nut_cap_screw_head_height_mm=1.35, nut_cap_screw_socket_af_mm=1.3,
    nut_cap_screw_socket_depth_mm=.75,
    nut_cap_tap_major_radius_mm=1., nut_cap_tap_pilot_radius_mm=.8,
    nut_cap_tap_major_bottom_z_mm=-21.5,
    nut_cap_tap_pilot_bottom_z_mm=-23.5,
    nut_cap_tap_entry_chamfer_mm=.2,
    foot_holes_xy_mm=[[-48., -81.], [6., -81.]],
    foot_top_z_mm=-24., foot_bottom_z_mm=-28.,
    foot_washer_top_z_mm=-23.1, foot_bolt_length_mm=20.,
    front_stop_y_mm=-84., rear_stop_y_mm=-96.,
    stop_z_span_mm=[-26., -24.5],
)

SOURCES = dict(
    M4_screw_25='https://www.accu.co.uk/metric-cap-head-screws/2830-SSC-M4-25-A2',
    M4_screw_20='https://www.accu.co.uk/metric-cap-head-screws/2828-SSC-M4-20-A2',
    M4_nut='https://www.accu.co.uk/hexagon-nuts/7890-HPN-M4-A2',
    M4_washer='https://www.accu.co.uk/metric-flat-washers/72488-HPW-M4-A2',
    M2_cap_screw='https://www.accu.co.uk/countersunk-socket-head-screws/5406-SSK-M2-6-A2',
    saved_M4='cad/thumb_ip_bench_v1/source_docs/README.md',
    saved_M2='cad/thumb_ip_drive_v1/source_docs/anchor_fastener/facts.json',
)


def build(cq, carrier_plate):
    """Construct only; the caller supplies and retains the unmodified input.

    Required input: motor plate in IP-local coordinates, four Y travel slots
    already cut by carrier.py, outside bounds X[-56,14],Y[-162.5,-92.5],
    Z[-21.8,-18.8]. No assembly is generated on import.

    returned ``parts`` does NOT repeat ``modified_carrier_plate``. Its expected
    scene name is drive_bench_baseplate, body carriage. ``machining_blanks`` are
    optional manufacturing-reference geometry, excluded from assembly/mass.
    ``required_common_holes`` asks the caller to add two M4 base stacks; this
    module does not silently modify the common plate or generate those stacks.
    """
    from OCP.BRepAlgoAPI import BRepAlgoAPI_Cut, BRepAlgoAPI_Fuse
    from OCP.BRepClass3d import BRepClass3d_SolidClassifier
    from OCP.TopAbs import TopAbs_OUT
    from OCP.TopTools import TopTools_ListOfShape

    p = dict(P)
    parts, contacts, assembly, extra, journal = [], [], [], [], []
    x, z = p['axis_x_mm'], p['axis_z_mm']
    axis = [x, p['screw_underhead_y_mm'], z]
    plate_name = 'drive_bench_baseplate'
    post_name = 'tension_adjuster_post'
    cap_name = 'tension_adjuster_head_cap'
    main_name = 'tension_adjuster_screw'
    washer_name = 'tension_adjuster_thrust_washer'
    nut_name = 'tension_adjuster_travelling_nut'
    lid_name = 'tension_adjuster_nut_lid'

    def facts(s):
        solids = s.Solids()
        bounded = []
        for solid in solids:
            cl = BRepClass3d_SolidClassifier(solid.wrapped)
            cl.PerformInfinitePoint(1e-7)
            bounded.append(cl.State() == TopAbs_OUT)
        return dict(valid=s.isValid(), solid_count=len(solids),
                    volume_mm3=sum(a.Volume() for a in solids),
                    positive_solids=bool(solids) and all(a.Volume() > 1e-6 for a in solids),
                    bounded=bool(solids) and all(bounded))

    def require(s, label):
        d = facts(s)
        if not (d['valid'] and d['solid_count'] == 1 and d['positive_solids'] and d['bounded']):
            raise ValueError(label+': '+str(d))
        return d

    def boolean(kind, left, *tools):
        before = [facts(v)['volume_mm3'] for v in (left, *tools)]
        op = {'cut': BRepAlgoAPI_Cut, 'fuse': BRepAlgoAPI_Fuse}[kind]()
        aa, bb = TopTools_ListOfShape(), TopTools_ListOfShape()
        aa.Append(left.copy().wrapped)
        for s in tools:
            bb.Append(s.copy().wrapped)
        op.SetArguments(aa); op.SetTools(bb)
        op.SetNonDestructive(True); op.SetRunParallel(False); op.Build()
        if not op.IsDone():
            raise ValueError('Incomplete '+kind)
        result = cq.Shape.cast(op.Shape())
        after = [facts(v)['volume_mm3'] for v in (left, *tools)]
        if any(abs(a-b) > 1e-8 for a, b in zip(before, after)):
            raise ValueError('Boolean mutated input')
        d = require(result, kind)
        if kind == 'cut' and d['volume_mm3'] > before[0]+1e-6:
            raise ValueError('Cut added material')
        journal.append(dict(operation=kind, input_volume_mm3=before[0], **d))
        return result

    def box(a, b):
        return cq.Solid.makeBox(b[0]-a[0], b[1]-a[1], b[2]-a[2], cq.Vector(*a))

    def cyl_y(xx, zz, r, y0, y1):
        return cq.Solid.makeCylinder(r, y1-y0, cq.Vector(xx, y0, zz), cq.Vector(0, 1, 0))

    def cyl_z(xx, yy, r, z0, z1):
        return cq.Solid.makeCylinder(r, z1-z0, cq.Vector(xx, yy, z0), cq.Vector(0, 0, 1))

    def hex_y(xx, zz, af, y0, y1):
        r = af/math.sqrt(3.)
        pts = [cq.Vector(xx+r*math.cos(i*math.pi/3), y0,
                         zz+r*math.sin(i*math.pi/3)) for i in range(6)]
        return cq.Solid.extrudeLinear(cq.Wire.makePolygon(pts, close=True), [], cq.Vector(0, y1-y0, 0))

    def hex_z(xx, yy, af, z0, z1):
        r = af/math.sqrt(3.)
        pts = [cq.Vector(xx+r*math.cos(i*math.pi/3), yy+r*math.sin(i*math.pi/3), z0) for i in range(6)]
        return cq.Solid.extrudeLinear(cq.Wire.makePolygon(pts, close=True), [], cq.Vector(0, 0, z1-z0))

    def socket_screw_y(xx, zz, underhead, length):
        s = boolean('fuse', cyl_y(xx, zz, 2., underhead-length, underhead),
                    cyl_y(xx, zz, 3.5, underhead, underhead+4.))
        return boolean('cut', s, hex_y(xx, zz, 3., underhead+2., underhead+4.01))

    def nut_y(xx, zz, lo, hi):
        return boolean('cut', hex_y(xx, zz, 7., lo, hi), cyl_y(xx, zz, 2., lo-.01, hi+.01))

    def add(name, shape, body, role, material='aluminum', color=(.25, .34, .42)):
        if any(row['name'] == name for row in parts):
            raise ValueError('Duplicate part '+name)
        row = dict(name=name, shape=shape, body=body, role=role, material=material,
                   color=list(color), geometry=require(shape, name))
        if body == 'adjuster_rotor':
            row.update(adjuster_axis=list(axis), adjuster_axis_direction=[0., 1., 0.],
                       travel_mm_per_revolution=p['signed_travel_mm_per_revolution'])
        parts.append(row)

    def seat(a, b, origin, normal, area, frame, text):
        contacts.append(dict(a=a, b=b, plane_origin_mm=origin, normal=normal,
                             minimum_area_mm2=area, contact_frame=frame, description=text))

    def path(part, translation, frame, excluded, stage, description):
        assembly.append(dict(kind='sampled_translation', part=part,
            initial_translation_mm=translation, final_translation_mm=[0., 0., 0.],
            maximum_step_mm=.5, stage_excluded=list(excluded), access_frame=frame,
            stage=stage, description=description))

    original = require(carrier_plate, 'carrier input')
    bb = carrier_plate.BoundingBox()
    if any(abs(a-b) > .02 for a, b in zip(
        (bb.xmin, bb.xmax, bb.ymin, bb.ymax, bb.zmin, bb.zmax),
        (-56., 14., -162.5, -92.5, -21.8, -18.8))):
        raise ValueError('Expected slotted original motor plate in IP-local mm')

    # Integral front tab: positive-volume Y overlap with the source plate.
    modified = boolean('fuse', carrier_plate, box(*p['tab_bounds_mm']))
    pocket = hex_y(x, z, p['nut_pocket_af_mm'], *p['nut_pocket_y_mm'])
    loading = box((x-4.2, -91.3, z), (x+4.2, -87.9, -16.99))
    # At s=+2 the fixed screw tip is at carrier-local Y=-100. This real
    # extension cuts an open relief in the thin original plate, not only the tab.
    # Its strength and proximity to existing motor fasteners remain unverified.
    modified = boolean('cut', modified, pocket, loading,
                       cyl_y(x, z, 2.25, -100.25, -85.99))
    # Drill blanks are distinct from the assembly's nominal M2 major bores.
    pilots = []
    for xx in p['nut_cap_screw_x_mm']:
        yy = p['nut_cap_screw_y_mm']
        pilot = cyl_z(xx, yy, .8, -23.5, -16.99)
        point_h = .8/math.tan(math.radians(59.))
        point = cq.Solid.makeCone(0., .8, point_h,
                    cq.Vector(xx, yy, -23.5-point_h), cq.Vector(0, 0, 1))
        pilot = boolean('fuse', pilot, point)
        pilots.append(pilot)
    machining_blank = boolean('cut', modified, *pilots)
    modified = machining_blank.copy()
    for xx in p['nut_cap_screw_x_mm']:
        yy = p['nut_cap_screw_y_mm']
        major = cyl_z(xx, yy, 1., -21.5, -16.99)
        entry = cq.Solid.makeCone(1., 1.2, .2,
                    cq.Vector(xx, yy, -17.2), cq.Vector(0, 0, 1))
        modified = boolean('cut', modified, major, entry)

    lid = box(*p['nut_cap_bounds_mm'])
    # Local relief means roof Z=-16.9, rather than clamping the hex nut vertically.
    lid = boolean('cut', lid, box((x-4.2, -91.3, -17.01), (x+4.2, -87.9, -16.9)))
    lid_screws = []
    for i, xx in enumerate(p['nut_cap_screw_x_mm'], 1):
        yy = p['nut_cap_screw_y_mm']
        cone = cq.Solid.makeCone(1., 2.35, 1.35,
                    cq.Vector(xx, yy, -16.35), cq.Vector(0, 0, 1))
        lid = boolean('cut', lid, cyl_z(xx, yy, 1.1, -17.01, -14.99), cone)
        screw = boolean('fuse', cyl_z(xx, yy, 1., -21., -16.35), cone)
        screw = boolean('cut', screw, hex_z(xx, yy, 1.3, -15.75, -14.99))
        name = 'tension_adjuster_nut_lid_screw_'+str(i)
        lid_screws.append(name)
        add(name, screw, 'carriage', 'Accu_SSK_M2_6_A2_countersunk_major_envelope', 'steel', (.65, .67, .70))
        assembly.append(dict(kind='countersunk_screw', part=name, top=[xx, yy, -15.],
            axis=[0., 0., 1.], shaft_radius_mm=1., head_radius_mm=2.35,
            head_height_mm=1.35, total_length_mm=6., length_includes_head=True,
            driver_radius_mm=.8, driver_length_mm=30., socket_across_flats_mm=1.3,
            entry_travel_mm=15., access_frame='carriage', stage='retain_travelling_nut',
            stage_excluded=[name], description='Countersunk screw, not a cylindrical-head underhead datum. Use actual saved shape for insertion; major thread bore is not a pilot drill.'))
        extra.append(dict(kind='conical_seat', a=name, b=lid_name,
            axis_origin_mm=[xx, yy, -16.35], axis_direction=[0., 0., 1.],
            radii_mm=[1.1, 2.35], finite_span_z_mm=[-16.25, -15.],
            minimum_area_mm2=18., frame='carriage',
            description='Actual truncated90-degree cone above the through-clearance intersection; require saved-shape face contact, not a zero-volume overlap claim.'))
    add(lid_name, lid, 'carriage', 'removable_nut_loading_lid')
    add(nut_name, nut_y(x, z, *p['nut_y_mm']), 'carriage',
        'Accu_HPN_M4_A2_keyed_travelling_nut_major_bore', 'steel', (.64, .66, .68))
    seat(lid_name, plate_name, [x, -89.6, -17.], [0., 0., 1.], 110., 'carriage',
         'Lid seats on real tab wings; the nut has a separate 0.1mm vertical capture gap.')

    # One connected fixed part with real feet, a thrust wall, open head chamber,
    # and two low integral end stops. No floating stop or detached support pads.
    post = box((-42., -81., -28.), (0., -73.9, -13.))
    post = boolean('fuse', post,
        box((-42., -81., -28.), (-28., -68.9, -13.)),
        box((-14., -81., -28.), (0., -68.9, -13.)),
        box((-55., -87., -28.), (-32., -74., -24.)),
        box((-10., -87., -28.), (13., -74., -24.)),
        box((-34., -84., -28.), (-8., -80.5, -24.5)),
        box((-34., -98., -28.), (-32., -80.5, -24.5)),
        box((-10., -98., -28.), (-8., -80.5, -24.5)),
        box((-34., -98., -28.), (-8., -96., -24.5)))
    post = boolean('cut', post, cyl_y(x, z, 2.25, -81.01, -73.89))
    for xx, yy in p['foot_holes_xy_mm']:
        post = boolean('cut', post, cyl_z(xx, yy, 2.25, -28.01, -23.99))
    cap = box((-42., -68.9, -28.), (0., -64.9, -13.))
    cap = boolean('cut', cap, cyl_y(x, z, p['front_cap_driver_radius_mm'], -68.91, -64.89))
    cap_screws, cap_nuts = [], []
    for i, xx in enumerate(p['cap_bolt_x_mm'], 1):
        # A real hex pocket provides counter-hold. The previous long rear
        # socket approach is deliberately not claimed: it hits the motor plate.
        post = boolean('fuse', post, box((xx-5., -84.2, -26.), (xx+5., -80.5, -15.)))
        post = boolean('cut', post, hex_y(xx, z, 7.2, -84.21, -81.))
        post = boolean('cut', post, cyl_y(xx, z, 2.25, -81.01, -68.89))
        cap = boolean('cut', cap, cyl_y(xx, z, 2.25, -68.91, -64.89))
        sn, nn = 'tension_adjuster_head_cap_screw_'+str(i), 'tension_adjuster_head_cap_nut_'+str(i)
        cap_screws.append(sn); cap_nuts.append(nn)
        add(sn, socket_screw_y(xx, z, -64.9, 20.), 'fixed',
            'Accu_SSC_M4_20_A2_cap_fastener_major_envelope', 'steel', (.65, .67, .70))
        add(nn, nut_y(xx, z, -84.2, -81.), 'fixed',
            'Accu_HPN_M4_A2_cap_fastener_major_bore', 'steel', (.65, .67, .70))
        seat(sn, cap_name, [xx, -64.9, z], [0., 1., 0.], 20., 'fixed', 'M4 cap head seats directly on machined cap, without washer.')
        seat(nn, post_name, [xx, -81., z], [0., 1., 0.], 24., 'fixed', 'M4 nut front face seats on post rear; nominal tip projects0.7mm.')
        assembly.append(dict(kind='axial_screw', part=sn, underhead=[xx, -64.9, z],
            axis=[0., 1., 0.], shaft_radius_mm=2., head_radius_mm=3.5,
            head_height_mm=4., length_mm=20., entry_travel_mm=25.,
            driver_radius_mm=2., driver_length_mm=40., socket_across_flats_mm=3.,
            access_frame='fixed', stage='fix_head_capture_cap', stage_excluded=[sn, nn],
            description='Insert from +Y and secure nut from -Y. Full module geometry and actual socket path remain to be audited.'))
        path(nn, [0., -5., 0.], 'fixed', [nn, sn], 'insert_keyed_cap_nut',
             'Insert actual nut5mm from -Y into keyed rear hex pocket; screw enters from+Y and turns against this pocket. No rear socket route is claimed. Temporary handling/holding until thread pickup remains unverified.')
        extra.append(dict(kind='keyed_nut_pocket', nut=nn, pocket_part=post_name,
            axis_origin_mm=[xx, -81., z], axis_direction=[0., 1., 0.],
            nut_af_mm=7., pocket_af_mm=7.2, finite_span_y_mm=[-84.2, -81.],
            minimum_nominal_wall_mm=.84, rear_open=True,
            description='Finite hex faces must provide torque reaction after fit clearance; no hand wrench or closed socket clearance is inherited.'))
    add(post_name, post, 'fixed', 'integral_thrust_post_feet_and_two_mechanical_stops')
    add(cap_name, cap, 'fixed', 'removable_head_capture_cap_with_AF3_tool_passage')
    main = socket_screw_y(x, z, -73., 25.)
    washer = boolean('cut', cyl_y(x, z, 4.5, -73.9, -73.), cyl_y(x, z, 2.15, -73.91, -72.99))
    add(main_name, main, 'adjuster_rotor', 'Accu_SSC_M4_25_A2_RH_adjuster_major_envelope', 'steel', (.70, .72, .74))
    add(washer_name, washer, 'adjuster_rotor', 'Accu_HPW_M4_A2_maximum_thickness_floating_thrust_washer', 'steel', (.70, .72, .74))
    seat(cap_name, post_name, [x, -68.9, z], [0., 1., 0.], 250., 'fixed', 'Cap seats on two actual post wings, not on the screw head.')
    seat(washer_name, post_name, [x, -73.9, z], [0., 1., 0.], 45., 'fixed', 'Nominal rear capture limit: washer/post thrust seat.')
    seat(main_name, washer_name, [x, -73., z], [0., 1., 0.], 22., 'fixed', 'Nominal rear capture limit: rotating head underside on separate washer.')

    # Assembly order is part of each declaration. No exclusions are collision
    # waivers: only known not-yet-installed parts may be absent at that stage.
    path(nut_name, [0., 0., 12.], 'carriage', [nut_name, lid_name, *lid_screws, main_name],
         'load_travelling_nut_before_lid', 'Drop nut through the real top opening before installing lid or threading the lead screw.')
    path(lid_name, [0., 0., 12.], 'carriage', [lid_name, *lid_screws],
         'retain_travelling_nut', 'Lower lid onto tab wings; do not clamp the floating nut roof.')
    path(washer_name, [0., 20., 0.], 'fixed', [washer_name, main_name, cap_name, *cap_screws, *cap_nuts],
         'load_head_capture_before_cap', 'Slip washer into open front chamber before screw and cap.')
    path(main_name, [0., 30., 0.], 'fixed', [main_name, cap_name, *cap_screws, *cap_nuts],
         'thread_adjuster_before_cap', 'Smooth major-envelope insertion is only a clearance screen; actual screw must rotate into the keyed nut, not be pushed through a thread.')
    path(cap_name, [0., 15., 0.], 'fixed', [cap_name, *cap_screws, *cap_nuts],
         'fix_head_capture_cap', 'Install cap after washer and threaded lead screw; then fit its two M4 screws/nuts.')
    assembly.append(dict(kind='driver_access', part=main_name, access_frame='fixed',
        stage='adjust_only_with_carrier_clamps_released', tool_origin_mm=[x, -64.9, z],
        tool_tip_mm=[x, -69., z],
        tool_axis=[0., 1., 0.], driver_radius_mm=2., driver_length_mm=50.,
        socket_across_flats_mm=3., passage_radius_mm=2.1, stage_excluded=[],
        socket_engagement_span_y_mm=[-71., -69.],
        description='Exterior shaft starts at the screw head front and passes through the full cap bore. AF3 hex engagement inside the head is a separate shape/fit check, not a radius2 cylinder. Carrier clamp locking, handle space and torque/force are unqualified.'))

    extra.append(dict(kind='axial_float', part=main_name, group_parts=[main_name, washer_name],
        offsets_mm=[[0., 0., 0.], [0., .1, 0.]], access_frame='fixed',
        seats=[dict(offset_mm=[0., 0., 0.], a=washer_name, b=post_name,
                    plane_origin_mm=[x, -73.9, z], normal=[0., 1., 0.], minimum_area_mm2=45.),
               dict(offset_mm=[0., .1, 0.], a=main_name, b=cap_name,
                    plane_origin_mm=[x, -68.9, z], normal=[0., 1., 0.], minimum_area_mm2=23.)],
        description='Two correlated-domain vertices only; include the third vertex with screw+.1 and washer0 below. No nominal zero-area front seat is accepted.'))
    extra.append(dict(kind='correlated_axial_float', group_parts=[main_name, washer_name],
        frame='fixed', vertices=[
            {main_name: [0., 0., 0.], washer_name: [0., 0., 0.]},
            {main_name: [0., .1, 0.], washer_name: [0., 0., 0.]},
            {main_name: [0., .1, 0.], washer_name: [0., .1, 0.]}],
        inequalities='0 <= washer_shift_y <= screw_shift_y <= 0.1 mm at CAD maximum hardware dimensions',
        description='Finite clearance and seats; not thread-flank, wear or continuous tolerance certification.'))
    extra.append(dict(kind='axial_float', part=nut_name,
        offsets_mm=[[0., 0., 0.], [0., .2, 0.]], access_frame='carriage',
        seats=[dict(offset_mm=[0., 0., 0.], a=nut_name, b=plate_name,
                    plane_origin_mm=[x, -91.3, z], normal=[0., 1., 0.], minimum_area_mm2=20.),
               dict(offset_mm=[0., .2, 0.], a=nut_name, b=plate_name,
                    plane_origin_mm=[x, -87.9, z], normal=[0., 1., 0.], minimum_area_mm2=20.)],
        description='Separate real opposing nut-pocket walls; keyed lower hex and removable roof bound lift/rotation, tolerance rocking remains unqualified.'))
    for s, yy in ((-2., -96.), (2., -84.)):
        extra.append(dict(kind='carriage_stop', carriage_y_mm=s,
            a=plate_name, b=post_name, plane_origin_mm=[x, yy, -25.25],
            normal=[0., 1., 0.], minimum_area_mm2=25.,
