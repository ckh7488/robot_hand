"""Inert, nominal split-saddle concept for a detached thumb-IP coupon.

build(cq, parent_local) returns parts/spec/contacts/foot_holes/assembly.
The input is the unchanged saved structural parent shell in IP-local mm.
Finite planar material faces, not mesh bounds, define each support pad.
Four separate jaws approach laterally; two M4 cross bolts clamp each pair.
Foot bottoms seat at Z=-28. The caller supplies a common base and foot screws.
No polymer strength, clamp torque, friction, tolerance or fabrication approval.
"""
from __future__ import annotations

import math

PARENT = 'thumb_segment2_segment_shell'
PARAMETERS = dict(stations_y_mm=[-31.048860184754197, -15.658523302549728], patch_half_y_mm=1.1,
    side_patch_z_mm=[-5.0, 5.0], lower_patch_abs_x_mm=[8.5, 11.0],
    pad_extrusion_mm=6.0, leg_abs_x_mm=[16.0, 22.0],
    leg_z_mm=[-24.0, 5.0], leg_half_y_mm=7.6, foot_abs_x_mm=[16.0, 34.0],
    foot_z_mm=[-28.0, -23.0], foot_half_y_mm=6.0,
    foot_hole_abs_x_mm=29.0, foot_hole_radius_mm=2.2,
    foot_slot_half_travel_mm=.6, clamp_axis_z_mm=-14.5, clamp_y_offset_mm=4.0,
    clamp_clearance_radius_mm=2.2, clamp_shaft_radius_mm=2.0,
    clamp_underhead_x_mm=-22.0, clamp_underhead_length_mm=50.0,
    clamp_head_radius_mm=3.5, clamp_head_length_mm=4.0,
    clamp_nut_start_x_mm=22.0, clamp_nut_thickness_mm=3.2,
    clamp_nut_across_flats_mm=7.0)


def _box(cq, lo, hi):
    return cq.Workplane('XY').box(*[hi[i]-lo[i] for i in range(3)]).translate(
        tuple((hi[i]+lo[i])/2 for i in range(3))).val()


def _cyl(cq, r, length, start, axis):
    return cq.Solid.makeCylinder(r, length, cq.Vector(*start), cq.Vector(*axis))


def _op(cq, kind, left, *right):
    from OCP.BRepAlgoAPI import BRepAlgoAPI_Cut, BRepAlgoAPI_Fuse, BRepAlgoAPI_Common
    from OCP.TopTools import TopTools_ListOfShape
    op = {'cut': BRepAlgoAPI_Cut, 'fuse': BRepAlgoAPI_Fuse, 'common': BRepAlgoAPI_Common}[kind]()
    args, tools = TopTools_ListOfShape(), TopTools_ListOfShape()
    args.Append(left.copy().wrapped)
    for part in right: tools.Append(part.copy().wrapped)
    op.SetArguments(args); op.SetTools(tools)
    op.SetNonDestructive(True); op.SetRunParallel(False); op.Build()
    if not op.IsDone(): raise ValueError('Incomplete '+kind)
    return cq.Shape.cast(op.Shape())


def describe(shape):
    from OCP.BRepClass3d import BRepClass3d_SolidClassifier
    from OCP.TopAbs import TopAbs_OUT
    states=[]
    for s in shape.Solids():
        c=BRepClass3d_SolidClassifier(s.wrapped); c.PerformInfinitePoint(1e-7)
        states.append(c.State()==TopAbs_OUT)
    return dict(valid=bool(shape.isValid()), solid_count=len(shape.Solids()),
        signed_volume_mm3=sum(s.Volume() for s in shape.Solids()),
        positive=all(s.Volume()>0 for s in shape.Solids()), bounded=all(states))


def _plane(face):
    from OCP.BRep import BRep_Tool
    from OCP.GeomLib import GeomLib_IsPlanarSurface
    check=GeomLib_IsPlanarSurface(BRep_Tool.Surface_s(face.wrapped),1e-5)
    if not check.IsPlanar(): return None
    plane=check.Plan(); p=plane.Location(); n=plane.Axis().Direction()
    return [p.X(),p.Y(),p.Z()],[n.X(),n.Y(),n.Z()]


def _patch(cq, parent, sign, y, kind):
    """Select the largest actual outer planar face within a bounded footprint."""
    p=PARAMETERS; dy=p['patch_half_y_mm']
    if kind=='side':
        lo,hi=([11,y-dy,-5],[15,y+dy,5]) if sign>0 else ([-15,y-dy,-5],[-11,y+dy,5])
        out=[sign,0,0]
    else:
        xa,xb=p['lower_patch_abs_x_mm']
        lo,hi=([xa,y-dy,-13],[xb,y+dy,-8]) if sign>0 else ([-xb,y-dy,-13],[-xa,y+dy,-8])
        out=[0,0,-1]
    window=_box(cq,lo,hi); candidates=[]
    for index,face in enumerate(parent.Faces()):
        plane=_plane(face)
        if plane is None: continue
        origin,normal=plane
        if kind=='side' and abs(normal[0])<.98: continue
        if kind=='lower' and abs(normal[2])<.98: continue
        actual_normal=face.normalAt().toTuple()
        if sum(a*b for a,b in zip(actual_normal,out))<.98: continue
        clipped=_op(cq,'common',face,window)
        for actual in clipped.Faces():
            if actual.Area()>1:
                candidates.append((actual.Area(),index,actual,origin,normal))
    if not candidates: raise ValueError('No actual structural '+kind+' patch at '+str((sign,y)))
    area,index,face,origin,normal=max(candidates,key=lambda v:v[0])
    if kind=='side' and area<15: raise ValueError('Side patch too small: '+str(area))
    if kind=='lower' and area<3: raise ValueError('Lower patch too small: '+str(area))
    if sum(a*b for a,b in zip(normal,out))<0: normal=[-n for n in normal]
    # Extrusion is outward in a fixed axis. On these taper planes that has a
    # positive component along the outward normal; it cannot enter the shell.
    vec=cq.Vector(*[v*p['pad_extrusion_mm'] for v in out])
    pad=cq.Solid.extrudeLinear(face.outerWire(),face.innerWires(),vec)
    return pad,dict(kind=kind,parent_native_face_index=index,area_mm2=area,
        plane_origin_mm=origin,normal=normal,patch_box_mm=[lo,hi],
        extrusion_vector_mm=list(vec.toTuple()),actual_face_bounds_mm=[
            [face.BoundingBox().xmin,face.BoundingBox().ymin,face.BoundingBox().zmin],
            [face.BoundingBox().xmax,face.BoundingBox().ymax,face.BoundingBox().zmax]])


def _part(name,shape,role,color,density):
    d=describe(shape)
    if not (d['valid'] and d['solid_count']==1 and d['positive'] and d['bounded']):
        raise ValueError('Unusable new part '+name+' '+str(d))
    return dict(name=name,shape=shape,body='fixed',role=role,color=color,
                density_kg_per_mm3=density,geometric_state=d)


def build(cq,parent_local):
    """Construct four jaws and two nominal M4x50 screw/nut pairs, no I/O."""
    p=PARAMETERS; before=describe(parent_local)
    if not (before['valid'] and before['positive'] and before['bounded'] and before['solid_count']==1):
        raise ValueError('The unmodified parent must be one valid bounded positive solid')
    parts=[]; contacts=[]; feet=[]; assembly=[]; patches=[]; hardware=[]
    for station,y in enumerate(p['stations_y_mm'],1):
        cy=y+p['clamp_y_offset_mm'];cz=p['clamp_axis_z_mm']
        for side,sign in [('left',-1),('right',1)]:
            name=f'bench_parent_saddle{station}_{side}'
            pad,side_patch=_patch(cq,parent_local,sign,y,'side')
            lower,lower_patch=_patch(cq,parent_local,sign,y,'lower')
            a,b=p['leg_abs_x_mm']; xlo,xhi=(a,b) if sign>0 else (-b,-a)
            leg=_box(cq,[xlo,y-p['leg_half_y_mm'],-24],[xhi,y+p['leg_half_y_mm'],5])
            # This lower arm stays outside |X|<8.5 and below Z=-15.5,
            # leaving both X=+/-4 open cord tracks independent of its material.
            armlo,armhi=(8.5,22) if sign>0 else (-22,-8.5)
            arm=_box(cq,[armlo,y-4,-20],[armhi,y+4,-15.5])
            a,b=p['foot_abs_x_mm']; xlo,xhi=(a,b) if sign>0 else (-b,-a)
            foot=_box(cq,[xlo,y-6,-28],[xhi,y+6,-23])
            jaw=_op(cq,'fuse',pad,lower,leg,arm,foot)
            jaw=_op(cq,'cut',jaw,_cyl(cq,2.2,70,(-35,cy,cz),(1,0,0)))
            hx=sign*p['foot_hole_abs_x_mm']; travel=p['foot_slot_half_travel_mm']
            slot=_op(cq,'fuse',_cyl(cq,2.2,9,(hx-travel,y,-30),(0,0,1)),
                _cyl(cq,2.2,9,(hx+travel,y,-30),(0,0,1)),
                _box(cq,[hx-travel,y-2.2,-30],[hx+travel,y+2.2,-21]))
            jaw=_op(cq,'cut',jaw,slot)
            parts.append(_part(name,jaw,'custom_split_saddle',(.35,.42,.50),2.70e-6))
            for patch in (side_patch,lower_patch):
                patch.update(part=name); patches.append(patch)
                contacts.append(dict(a=name,b=PARENT,minimum_area_mm2=patch['area_mm2']*.999,
                    description='Actual finite structural '+patch['kind']+' face; nominal zero-gap contact, no interference fit',**patch))
            feet.append(dict(part=name,contact_part=name,part_name=name,xy=[hx,y],underhead_z=-22.1,
                bolt_length_mm=20.,nominal_bolt='M4x20, caller-supplied',hole_radius_mm=2.2,
                slot_half_travel_x_mm=travel,foot_span_z_mm=[-28,-23],
                base_contact_z_mm=-28,base_expected_span_z_mm=[-36,-28],
                insertion_axis=[0,0,-1],head_nominal_radius_mm=3.5,
                head_height_mm=4.,top_washer_z_mm=[-23,-22.1],
                bottom_washer_z_mm=[-36.9,-36],nut_expected_z_mm=[-40.1,-36.9],
                washer_thickness_for_stack_mm=.9,washer_thickness_tolerance_range_mm=[.7,.9],
                tip_z_mm=-42.1,nominal_tip_beyond_nut_mm=2.0))
            assembly.append(dict(kind='lateral_jaw',part=name,initial_translation_mm=[sign*15,0,0],
                final_translation_mm=[0,0,0],path='Translate inward along X before cross-bolt tightening; no rotation',
                validation='Swept translation requires the probe; all-coupon hardware/tool clearance belongs to integration'))
        bolt_name=f'bench_parent_saddle{station}_clamp_screw'
        nut_name=f'bench_parent_saddle{station}_clamp_nut'
        shaft=_cyl(cq,2,50,(-22,cy,cz),(1,0,0))
        head=_cyl(cq,3.5,4,(-26,cy,cz),(1,0,0))
        screw=_op(cq,'fuse',shaft,head)
        # Hex socket: 3mm across flats, 2mm deep, opened on the left head face.
        rc=3/math.sqrt(3)
        pts=[cq.Vector(-26.01,cy+rc*math.cos(k*math.pi/3),cz+rc*math.sin(k*math.pi/3)) for k in range(6)]
        socket=cq.Solid.extrudeLinear(cq.Wire.makePolygon(pts,close=True),[],cq.Vector(2.01,0,0))
        screw=_op(cq,'cut',screw,socket)
        rc=7/math.sqrt(3)
        pts=[cq.Vector(22,cy+rc*math.cos(k*math.pi/3),cz+rc*math.sin(k*math.pi/3)) for k in range(6)]
        nut=cq.Solid.extrudeLinear(cq.Wire.makePolygon(pts,close=True),[],cq.Vector(3.2,0,0))
        nut=_op(cq,'cut',nut,_cyl(cq,2,5,(21,cy,cz),(1,0,0)))
        parts.append(_part(bolt_name,screw,'nominal_M4x50_socket_screw',(.55,.58,.62),7.85e-6))
        parts.append(_part(nut_name,nut,'nominal_M4_hex_nut',(.55,.58,.62),7.85e-6))
        not_yet_installed=[f'bench_parent_mount_{i}_{suffix}'
            for i in range(1,5) for suffix in ('screw','upper_washer')]
        assembly.append(dict(kind='axial_screw',part=bolt_name,underhead=[-22,cy,cz],
            axis=[-1.,0.,0.],shaft_radius_mm=2.,head_radius_mm=3.5,head_height_mm=4.,
            length_mm=50.,entry_travel_mm=55.,driver_radius_mm=2.,driver_length_mm=40.,
            socket_across_flats_mm=3.,stage_excluded=[bolt_name,nut_name,*not_yet_installed],
            description='Clamp detached coupon with saddle foot screws/upper washers not yet installed; shaft enters along +X, driver approaches from -X. Requires fresh scene audit.'))
        assembly.append(dict(kind='axial_nut',part=nut_name,
            upper_face=[22,cy,cz],axis=[1.,0.,0.],across_flats_mm=7.,height_mm=3.2,
            mate_screw=bolt_name,socket_max_dia_mm=11.8,socket_length_mm=50.,
            path_travel_mm=30.,entry_travel_mm=30.,socket_part_number='05004504001',
            stage='before_parent_foot_fasteners',
            socket_reference='Wera 05004504001 official outer envelope; actual internal engagement not modeled',
            stage_excluded=[nut_name,bolt_name,*not_yet_installed],
            description='Nut seats at X=22; approach from +X before foot screws/upper washers are installed. Declared Wera external envelope is a pending access screen, not an installation pass.'))
        hardware.append(dict(screw=bolt_name,nut=nut_name,axis_origin_mm=[0,cy,cz],axis_direction=[1,0,0],
            shaft_span_x_mm=[-22,28],shaft_radius_mm=2,clearance_radius_mm=2.2,
            head_span_x_mm=[-26,-22],head_radius_mm=3.5,nut_span_x_mm=[22,25.2],
            simplified_thread_engagement_mm=3.2,tip_beyond_nut_mm=2.8,
            driver='3mm hex nominal; approach along +X from negative X',
            wrench='7mm across flats nominal nut; approach must be screened with an actual selected tool',
            thread_representation='Major-diameter solid/bore only, no helix or preload'))
        contacts.extend([dict(a=bolt_name,b=f'bench_parent_saddle{station}_left',minimum_area_mm2=20,
            description='Screw head underside against left leg X=-22',plane_origin_mm=[-22,cy,cz],normal=[1,0,0],patch_radius_mm=3.5),
            dict(a=nut_name,b=f'bench_parent_saddle{station}_right',minimum_area_mm2=20,
            description='Nut face against right leg X=22',plane_origin_mm=[22,cy,cz],normal=[1,0,0],patch_radius_mm=4.05)])
    if describe(parent_local)!=before: raise ValueError('Input parent was mutated')
    return dict(parts=parts,contacts=contacts,foot_holes=feet,assembly=assembly,
        spec=dict(units='mm',frame='IP-local',parameters=p,source_parent=PARENT,
            parent_unmodified=True,contact_patches=patches,clamp_hardware=hardware,
            base_top_z_mm=-28,base_complete=False,detached_coupon_only=True,
            assembly_order=['Remove coupon from upstream hinge; retain the declared IP assembly.',
                'Place four open jaws laterally onto the two measured shell reinforcement bands.',
                'Insert and tighten the two cross screws/nuts while parent foot screws/upper washers are absent.',
                'Seat the clamped coupon/fixture on the common base, then install foot fasteners through the X-adjustment slots.',
                'Route cords and install skin only in the independently checked sequence; do not infer whole-hand installation.'],
            part_count=len(parts),manufacturing_approved=False,
            limits=['Nominal machined jaw faces; no fit or manufacturing tolerance selected.',
                'Lower pads give a geometric downward load path. Uplift and Y sliding rely on clamp friction, not an independently verified positive lock.',
                'Clamp torque, polymer crush, stiffness, friction, strength and fatigue are unverified.',
                'Common base and four foot M4 screw/nut pairs are supplied by the integrating builder.',
                'Actual supply screw/nut tolerances and driver/wrench geometry are not established.',
                'No faceplate, pin, bushing or cord-slot edge is an intended clamp surface.',
                'Rope and full coupon motion against this new fixture require fresh integration checks.']))
